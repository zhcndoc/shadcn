## 用法

```tsx showLineNumbers
import {
  getLocalTimeZone,
  today,
  type CalendarDate,
} from "@internationalized/date"

import { Calendar } from "@/components/ui/calendar"
```

```tsx showLineNumbers
const [date, setDate] = React.useState<CalendarDate | null>(
  today(getLocalTimeZone())
)

return (
  <Calendar value={date} onChange={setDate} className="rounded-lg border" />
)
```

如需了解更多信息，请参阅 [React Aria](https://react-aria.adobe.com/Calendar) 文档。

## 自定义单元格大小

<ComponentPreview
  styleName="aria-nova"
  name="calendar-custom-days"
  title="自定义单元格大小"
  description="具有自定义单元格大小且可响应式调整的日历。"
  className="**:[.preview]:h-[560px]"
/>

你可以使用 `--cell-size` CSS 变量自定义日历单元格的大小。还可以通过使用特定断点的值使其具有响应式效果：

```tsx showLineNumbers
<Calendar
  value={date}
  onChange={setDate}
  className="rounded-lg border [--cell-size:--spacing(11)] md:[--cell-size:--spacing(12)]"
/>
```

或者使用固定值：

```tsx showLineNumbers
<Calendar
  value={date}
  onChange={setDate}
  className="rounded-lg border [--cell-size:2.75rem] md:[--cell-size:3rem]"
/>
```
